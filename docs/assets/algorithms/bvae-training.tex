\begin{algorithm}[t]
\caption{Training the Hypergraph $\beta$-VAE}
\label{alg:bvae_train}
\DontPrintSemicolon

\KwIn{Sub-hypergraphs $(X^V,X^E,H)$, KL weights $\beta_V,\beta_E$, flag \textsc{Unconstrained}}
\KwOut{Trained $\beta$-VAE encoder and decoder}

Initialize the node encoder, hyperedge encoder, incidence decoder, and, if needed, node decoder\;

\For{each training epoch}{
    \For{each mini-batch $(X^V,X^E,H)$}{
        Encode nodes and hyperedges into latent variables $Z^V$ and $Z^E$\;

        Reconstruct the incidence matrix from $(Z^V,Z^E)$\;

        Compute the incidence reconstruction loss and KL regularization losses\;

        \If{\textsc{Unconstrained}}{
            Decode node representations for entity resolution\;

            Compute the node consistency loss across overlapping sub-hypergraphs\;

            \If{this is an incidence-reconstruction step}{
                Backpropagate only the incidence reconstruction objective\;
            }
            \Else{
                Backpropagate only the node consistency objective\;
            }
        }
        \Else{
            Backpropagate the incidence reconstruction objective\;
        }

        Update the trainable parameters\;
    }

    Evaluate the validation loss\;
}

Select the checkpoint with the best validation loss\;

\Return{trained encoder and decoder}
\end{algorithm}
