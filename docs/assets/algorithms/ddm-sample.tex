\begin{algorithm}[t]
\caption{Sampling with the Hypergraph Latent Diffusion Model}
\label{alg:ldm_sample}
\DontPrintSemicolon

\KwIn{Trained latent diffusion model, pretrained $\beta$-VAE decoder, number of reverse steps $T$, flag \textsc{Unconstrained}}
\KwOut{Generated sub-hypergraph}

\eIf{\textsc{Unconstrained}}{
    Initialize node latents $Z_T^V$ with Gaussian noise\;
}{
    Take the input node features and map them to fixed node latents $Z^V$\;
}

Initialize hyperedge latents $Z_T^E$ with Gaussian noise\;

\For{$t = T, T-1, \dots, 1$}{
    \If{\textsc{Unconstrained}}{
        Update the node latents by one reverse diffusion step, conditioned on the current hyperedge latents\;
    }

    Update the hyperedge latents by one reverse diffusion step, conditioned on the current node latents\;
}

Decode the final latents $(Z_0^V, Z_0^E)$ into incidence logits using the pretrained $\beta$-VAE decoder\;

Sample a discrete incidence matrix from the decoded logits\;

\If{\textsc{Unconstrained}}{
    Decode node representations for entity resolution\;
}

Build the generated sub-hypergraph from the sampled incidence matrix\;

\Return{generated sub-hypergraph}
\end{algorithm}
